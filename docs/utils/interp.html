def lagrange(x, xp, yp):
    n = len(xp)
    # def get_basis(X, j):
    #     b = [
    #         (X - xp[m]) / (xp[j] - xp[m])
    #         for m in range(n) if not m == j
    #     ]
    #     return np.prod(b, axis=0) * yp[j]
    # return np.sum([get_basis(x, j) for j in range(n)], axis=0)

    basis = np.zeros((len(x), len(x)), dtype=yp.dtype)
    for j in range(n):
        b = []
        for m in range(n):
            if m == j:
                continue
            b.append((x - xp[m]) / (xp[j] - xp[m]))
        basis[j,:] = np.prod(b, axis=0) * yp[j]
    return np.sum(basis, axis=0)

    q(x) = (1-t(x))y1 + t(x)y2 + t(x)(1-t(x))((1- t(x))a+t(x)b)
    t(x) = (x-x1)/(x2-x1)
    a = k1(x2-x1)-(y2-y1)
    b = -k2(x2-x1)+y2-y1
    [
        [ a11 a12 0   ] [k0] = [b1]
        [ a21 a22 a23 ] [k1] = [b2]
        [ 0   a32 a33 ] [k2] = [b3]
    ]
    a11 = 2/(x1-x0)
    a12 = 1/(x1-x0)
    a21 = 1/(x1-x0)

    a23 = 1/(x2-x1)
    a32 = 1/(x2-x1)
    a33 = 2/(x2-x1)
    b1 = 3*( (y1-y0)/(x1-x0)**2 )
    b2 = 3*( (y1-y0)/(x1-x0)**2 + ( (y2-y1)/(x2-x1**2) ) )
    b3 = 3*( (y2-y1)/(x2-x1)**2 )
    """
    xp = [-1, 0, 3]
    yp = [0.5, 0, 3]

    px = xp[0:3]

    a11 = 2/(px[1]-px[0])
    a12 = 1/(px[1]-px[0])

    a_mat = [
